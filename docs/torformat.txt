// EA Mythic file format, used for storing assets in SWTOR (.tor)
// also used in Ultima Online (.uop) and Warhammer Online (.myp)

struct HEADER {
  uint32 magic // always 0x0050594D = "MYP\0"
  uint32 version // either 5 (Ultima, Warhammer, SWTOR 32-bit client) or 6 (SWTOR 64-bit client)
  uint32 byteOrder // always 0xFD23EC43 (little endian) in .tor but can be 0x43EC23FD (big endian)
  uint64 ftOffset
  uint32 ftCapacity // always 1000
  uint32 totalNumberOfFiles // including metadata.bin, ft.sig and (sometimes) groupmanifest.bin
  uint32 confirmSequence // number of 500MB blocks in the file
  uint32 writeSequence // identical to confirm sequence
  // read first FILE_TABLE at ftOffset
}

struct FILE_TABLE {
  uint32 ftCapacity;
  uint64 ftNewOffset; // offset of next file table, or zero if this was the last file table
  for each (ftCapacity) {
    uint64 offset, or zero if empty
    uint32 metadata length // metadata is stored at offset, right before file. file starts at offset + metadata length
    uint32 stored length
    uint32 uncompressed length
    uint32 primary hash // two-part file name hash - according to Warhammer's mft.myp, ph is before sh
    uint32 secondary hash
    uint32 metadata checksum // hashlittle2 of metadata (02 00 20 00 XX ... XX). Doesn't appear to ever be checked by client
    uint16 isCompressed - compression type (ct), 0 = uncompressed, 1 = compressed (DEFLATE in v5, zstd in v6)
  }
  // read next FILE_TABLE starting at ftNewOffset
}

struct FILE {
  // <-- starting at offset
  // metadata is optional
  metadata {
      uint16 always 2 (metadata version? metadata bitset?)
      uint16 payload length, always 32 (0x20)
      char[32] SHA-256 checksum of uncompressed file contents
  }
  char[compressed length (cl)] file contents, either compressed or uncompressed
}
